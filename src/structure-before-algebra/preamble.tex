% tier chips
\makeatletter
\newcommand{\tierchip}[1]{%
  \begingroup
  \sffamily\bfseries\fontsize{6.4}{7}\selectfont
  \setlength{\fboxsep}{2.4pt}%
  \tb@tierchip{#1}%
  \endgroup}
\def\tb@tierchip#1{%
  \def\tb@ta{#1}%
  \def\tb@ttE{easy}\def\tb@ttM{medium}\def\tb@ttH{hard}\def\tb@ttC{core}%
  \ifx\tb@ta\tb@ttE\colorbox{emerald!15}{\textcolor{emerald!85!ink}{EASY}}\else
  \ifx\tb@ta\tb@ttM\colorbox{gold!20}{\textcolor{gold!72!ink}{MEDIUM}}\else
  \ifx\tb@ta\tb@ttH\colorbox{coral!16}{\textcolor{coral!82!ink}{HARD}}\else
  \ifx\tb@ta\tb@ttC\colorbox{cyan!15}{\textcolor{cyan!82!ink}{CORE}}\else
  \colorbox{plum!15}{\textcolor{plum!82!ink}{\MakeUppercase{#1}}}\fi\fi\fi\fi}
\makeatother

% \tech{Name}{tier}
\makeatletter
\newcommand{\tech}[3][NOKEY]{%
  \par\needspace{4.2\baselineskip}%
  \vspace{7pt}%
  \refstepcounter{tech}%
  \phantomsection
  \def\tb@key{#1}\def\tb@nokey{NOKEY}%
  \ifx\tb@key\tb@nokey\index{#2}\else\index{#1}\fi
  \noindent\parbox[t]{\dimexpr\textwidth-0.95in\relax}{%
    \raggedright\sffamily\bfseries\color{navy}\fontsize{10.4}{12.4}\selectfont
    {\color{emerald}\thetech}\hspace{0.45em}#2}%
  \hfill\raisebox{0.12\baselineskip}{\tierchip{#3}}%
  \par\nobreak\vspace{2.4pt}%
  \nobreak}
\makeatother
